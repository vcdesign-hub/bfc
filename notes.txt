download image links
------------------------

$urls = Get-Content images.txt
$names = Get-Content names.txt

if ($urls.Count -ne $names.Count) {
    Write-Host "Mismatch! URLs: $($urls.Count), Names: $($names.Count)"
    exit
}

$nameCount = @{}

for ($i = 0; $i -lt $urls.Count; $i++) {
    try {
        $url = $urls[$i]
        $baseName = $names[$i].Trim()

        # Track duplicates
        if ($nameCount.ContainsKey($baseName)) {
            $nameCount[$baseName]++
        } else {
            $nameCount[$baseName] = 1
        }

        $count = $nameCount[$baseName]

        # Get extension from URL
        $ext = [System.IO.Path]::GetExtension($url)

        # Build filename
        if ($count -eq 1) {
            $filename = "$baseName$ext"
        } else {
            $filename = "$baseName-$count$ext"
        }

        Write-Host "Downloading -> $filename"

        Invoke-WebRequest $url -OutFile $filename -ErrorAction Stop

    } catch {
        Write-Host "Failed: $url"
    }
}
